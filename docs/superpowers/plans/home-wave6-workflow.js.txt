const results = []
const WANT = Array.isArray(args) && args.length ? args : Object.keys(TASKS)
const pending = Object.keys(TASKS).filter((k) => WANT.includes(k))
const state = {}
for (const k of Object.keys(TASKS)) state[k] = WANT.includes(k) ? 'pending' : 'done'
const LIMIT = 8
const running = new Map()

function ready(name) {
  return (TASKS[name].after || []).every((d) => state[d] === 'done')
}
function blocked(name) {
  return (TASKS[name].after || []).some((d) => state[d] === 'failed' || state[d] === 'skipped')
}

async function run(name) {
  state[name] = 'running'
  log(`start ${name}`)
  const r = await agent(`${COMMON}\n${TASKS[name].prompt}`, { label: name, phase: 'Wave 6', schema: RESULT })
  state[name] = r ? 'done' : 'failed'
  results.push({ name, ...(r || { summary: 'NO RESULT (agent failed or was skipped)' }) })
  log(`${state[name]} ${name}`)
}

phase('Wave 6')
while (pending.length || running.size) {
  for (const name of pending.slice()) {
    if (blocked(name)) {
      state[name] = 'skipped'
      pending.splice(pending.indexOf(name), 1)
      log(`skip ${name}: a task it needs did not finish`)
      results.push({ name, summary: 'SKIPPED: a task it needs did not finish' })
      continue
    }
    if (running.size >= LIMIT) break
    if (ready(name)) {
      pending.splice(pending.indexOf(name), 1)
      const p = run(name).then(() => running.delete(name))
      running.set(name, p)
    }
  }
  if (running.size === 0) break
  await Promise.race([...running.values()])
}
return results
