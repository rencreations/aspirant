const queue = Object.keys(TASKS).filter((k) => WANT.includes(k))
const LIMIT = 9

async function worker() {
  while (queue.length) {
    const name = queue.shift()
    log(`start ${name}`)
    const r = await agent(`${COMMON}\n${TASKS[name].prompt}`, { label: name, phase: 'Wave 7', schema: RESULT })
    results.push({ name, ...(r || { summary: 'NO RESULT (agent failed or was skipped)' }) })
    log(`done ${name}`)
  }
}

phase('Wave 7')
await Promise.all(Array.from({ length: LIMIT }, () => worker()))
