const results = []
const queue = Object.keys(TASKS).filter((k) => !Array.isArray(args) || !args.length || args.includes(k))
const LIMIT = 8

async function worker() {
  while (queue.length) {
    const name = queue.shift()
    log(`start ${name}`)
    const r = await agent(`${COMMON}\n${TASKS[name].prompt}`, { label: name, phase: 'Walk and fix', schema: RESULT })
    results.push({ name, ...(r || { summary: 'NO RESULT (agent failed or was skipped)' }) })
    log(`done ${name}`)
  }
}

phase('Walk and fix')
await Promise.all(Array.from({ length: LIMIT }, () => worker()))
return results
