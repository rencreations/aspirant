  while (queue.length) {
    const name = queue.shift()
    log(`start ${name}`)
    const r = await agent(`${COMMON}\n${TASKS[name].prompt}`, { label: name, phase: 'Wave 12', schema: RESULT })
    results.push({ name, ...(r || { summary: 'NO RESULT (agent failed or was skipped)' }) })
    log(`done ${name}`)
  }
}
